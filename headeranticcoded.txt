  <script>
    document.addEventListener("DOMContentLoaded", function () {
      const menuActivador = document.getElementById("menuActivador");
      const popup = document.getElementById("popup");
      const menuActivadorInPopup = document.getElementById(
        "menuActivadorInPopup"
      );

      // Mostrar el pop-up después de la animación del menú activador
      menuActivador.addEventListener("click", function () {
        menuActivador.classList.add("open");
        setTimeout(function () {
          popup.style.display = "flex";
          popup.classList.add("show");
          menuActivador.style.display = "none";
          menuActivadorInPopup.style.display = "flex";
        }, 300);
      });

      // Cerrar el pop-up al hacer clic en el menú activador dentro del pop-up
      menuActivadorInPopup.addEventListener("click", function () {
        popup.style.display = "none";
        popup.classList.remove("show");
        menuActivadorInPopup.style.display = "none";
        menuActivador.style.display = "flex";
        setTimeout(function () {
          menuActivador.classList.remove("open");
        }, 10);
      });
    });
  </script>
